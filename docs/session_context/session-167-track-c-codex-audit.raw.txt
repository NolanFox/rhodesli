    49	def _slugify(name: str) -> str:
    50	    """Lowercase, hyphenated, URL-safe slug from a display name.
    51	
    52	    Strips to ascii alnum + single hyphens, collapses runs, trims edges.
    53	    Returns "" if nothing usable remains (caller must handle)."""

    62	def _dedupe_slug(base_slug: str, existing_slugs: set) -> str:
    63	    """Return base_slug, or base_slug-<n> if taken. Caps suffix scan; falls
    64	    back to a random suffix so we never loop forever."""
    65	    if base_slug and base_slug not in existing_slugs:
    66	        return base_slug
    67	    for n in range(2, 50):
    68	        candidate = f"{base_slug}-{n}"
    69	        if candidate not in existing_slugs:
    70	            return candidate
    71	    # Pathological collision — append short random hex.
    72	    import secrets
    73	
    74	    return f"{base_slug}-{secrets.token_hex(3)}"

   245	def _check_origin(request) -> StarletteResponse | None:
   246	    """Validate that a POST request comes from an allowed origin.
   247	
   248	    Checks the Origin header first, falls back to Referer.
   249	    Returns None if the origin is allowed or absent (same-origin requests
   250	    may omit both headers). Returns a 403 Response if the origin doesn't
   251	    match any allowed domain.
   252	    """
   253	    origin = request.headers.get("origin") or ""
   254	    if not origin:
   255	        # Fall back to Referer
   256	        origin = request.headers.get("referer") or ""
   257	    if not origin:
   258	        # No Origin or Referer — likely a same-origin request; allow
   259	        return None
   260	
   261	    try:
   262	        parsed = urlparse(origin)
   263	        hostname = parsed.hostname or ""
   264	    except Exception:
   265	        return StarletteResponse("Forbidden — invalid origin", status_code=403)
   266	
   267	    if hostname in _ALLOWED_ORIGINS:
   268	        return None
   269	
   270	    return StarletteResponse("Forbidden — origin not allowed", status_code=403)
